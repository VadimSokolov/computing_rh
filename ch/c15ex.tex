\section{Exercises}\label{chap:phase:ex}

\paragraph{On Sections~\ref{cp:sec:transfer} and~\ref{cp:sec:phase}.}
\begin{exercises}
\item\label{ex:cp:cauchy} Prove the semigroup identity $P_a*P_b=P_{a+b}$ from the characteristic function $e^{-a|\xi|}$ of $P_a$, and deduce the identity $\int_\R P_d(\theta-\gamma)(\theta^2+u)^{-1}d\theta=\pi P_{d+\sqrt u}(\gamma)/\sqrt u$ used in Lemma~\ref{cp:lem:pair}.
\item Show that the Thorin density of Lemma~\ref{cp:lem:pair}(a), in the chart $z=\theta^2$, has total mass one, and that it converges vaguely to $\delta_{\gamma^2}$ as $d\downarrow0$.
\item\label{ex:cp:level} Show that the level sets of $t+2(\alpha-\frac12)^2$ are the parabolas along which Theorem~\ref{cp:thm:phase}(i) carries the boundary of a certified region, and that the parabola through $(t_0,\alpha_0)$ meets the axis $\alpha=\frac12$ at $t_0+2(\alpha_0-\frac12)^2$.
\item Using Theorem~\ref{cp:thm:phase}(ii) and Corollary~\ref{cp:cor:known}(d), find the $\alpha$ for which a zero free half plane $\re s>\alpha$ at time zero would improve the bound $\Lambda\le0.2$, and the smallest $t$ for which the flowed half plane $\re s>\frac78$ gives a point of $\mathcal G$ at the basepoint $0.6$.
\item Let $F$ be the perturbation of Theorem~\ref{thm:ch13:perturb}, with a quadruple of zeros at $\frac12\pm\frac14\pm ig$. Assuming RH, show that the shifted clock of $F$ at the basepoint $\alpha$ is a GGC exactly for $\alpha\ge\frac34$, and compute its Thorin density near $\theta=g$ for $\alpha=0.8$.
\end{exercises}

\paragraph{On Sections~\ref{cp:sec:see} and~\ref{cp:sec:cert}.}
\begin{exercises}
\item\label{ex:cp:mean} Assuming RH, show that the average of $v_0(\alpha,\theta)$ over a window of length $H$ about $\theta$ is $\frac12\log(\theta/2\pi)+O(\log\theta/H)$ when $H$ is large compared with $\alpha-\frac12$, and explain why the minimum over the window can be much smaller than the average when $\alpha-\frac12$ is small compared with the mean spacing.
\item Show that $v_t(\alpha,0)=\xi_t'/\xi_t(\alpha)$ is positive for every $t\ge0$ and $\alpha>\frac12$, using Lemma~\ref{cp:lem:structure}(ii), and compute it at $t=0.2$, $\alpha=0.6$ from the P\'olya integral.
\item\label{ex:cp:grid} Prove Lemma~\ref{cp:lem:grid} and show that it remains true with $\le$ in place of $<$, the change of argument then lying in $(-\frac\pi2,\frac\pi2)$.
\item Explain why the number of grid points in Theorem~\ref{cp:thm:cert} grows like $e^{(\pi/4-2\eta)\theta}$ for a tilt $\eta<\pi/8$.
\item Show that positivity of $v_t$ on the boundary of a rectangle does not by itself exclude a zero inside it, and explain why Proposition~\ref{cp:prop:thorincert} asks for the whole rectangle.
\end{exercises}

\paragraph{On Section~\ref{cp:sec:uniform}.}
\begin{exercises}
\item Show that for $g$ supported in $[-L/2,L/2]$ the function $h=g*\tilde g$ is supported in $[-L,L]$, and that the prime sum in Weil's explicit formula runs over $n\le e^L$.
\item Derive the $x$ space form of the archimedean term of Section~\ref{cp:sec:weil} from the integral representation $\psi(z)=\int_0^\infty(e^{-t}/t-e^{-zt}/(1-e^{-t}))\,dt$.
\item For an isolated pair of zeros of $H_t$ at distance $D$, show from the equation of motion that $D^2\approx D_0^2+8t$ (compare Exercises~\ref{ex:ch21:motion} and~\ref{ex:ch21:pair}). Explain why two certified sign changes of $\xi_t(\frac12+i\theta)$ on a grid of step $0.01$ show that the pair of Proposition~\ref{cp:prop:gas}(iii) is still real at $t$, while their absence does not show that it has left the line, and why the lower end of the bracket there rests on the law $D^2\approx8(t-t_*)$.
\item Fit $\Lambda=c/\log X$ to the thirteen points of Proposition~\ref{cp:prop:cost} by least squares in $1/\log X$ and check the value $c=5.59$; then fit $\Lambda=c'\log\log X/\log X$ to the same points and compare the barrier each law gives for $\Lambda=0.1$ with the last row of the table, $X\approx9\times10^{21}$.
\end{exercises}
